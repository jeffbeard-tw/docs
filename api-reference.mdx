export const ApiTile = ({ method, title, path, href }) => {
  const kind = method === "GET" ? "api-tile-get" : method === "POST" ? "api-tile-post" : "api-tile-other";
  return (
    <a href={href} className={"api-tile " + kind}>
      <div className="flex items-center gap-2">
        <span className="text-xs font-bold">{method}</span>
        <span className="font-semibold">{title}</span>
      </div>
      <div className="mt-2 text-sm break-all" style={{ fontFamily: "monospace" }}>{path}</div>
    </a>
  );
};
