export const ApiTile = ({ method, title, href }) => {
  const kind = method === "GET" ? "api-tile-get" : method === "POST" ? "api-tile-post" : method === "PATCH" ? "api-tile-patch" : "api-tile-other";
  return (
    <a href={href} className={"api-tile " + kind}>
      <div className="flex items-center justify-center gap-2 text-sm">
        <span className="text-xs font-bold">{method}</span>
        <span className="font-semibold">{title}</span>
      </div>
    </a>
  );
};
